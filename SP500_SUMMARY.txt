HOW TO USE
================================================================================

QUICK START (3 MINUTES):
  1. python -m venv venv
  2. source venv/bin/activate  (or: venv\Scripts\activate on Windows)
  3. pip install -r requirements.txt
  4. export MASSIVE_COM_API_KEY="your_key"
  5. python main.py AAPL MSFT GOOGL

ANALYZE DIFFERENT STOCKS:
  python main.py NVDA AMD INTEL           # Tech stocks
  python main.py JPM BAC GS               # Finance stocks
  python main.py JNJ UNH MRK              # Healthcare stocks
  python main.py WMT HD COST              # Consumer stocks

USE RANDOM SAMPLES:
  python main.py --sample 10              # 10 random
  python main.py --sample 25              # 25 random
  python main.py --sample 50              # 50 random

GET S&P 500 INFO:
  python main.py --list-sp500             # All 500 tickers
  python main.py --count-sp500            # Total count
  python main.py --validate AAPL X MSFT   # Validate symbols

PYTHON EXAMPLES:
  from main import StockOptionsMLAgent
  agent = StockOptionsMLAgent()
  results = agent.run(["AAPL", "MSFT"])
  # or
  results = agent.run(use_sample=True, sample_size=20)
